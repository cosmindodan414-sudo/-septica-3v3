socket.on('disconnect', () => { for (const room of rooms.values()) { const seat = room.players.findIndex( p => p.socketId === socket.id ); if (seat === -1) { continue; } const player = room.players[seat]; // Dacă jocul NU a început, // scoatem jucătorul complet din cameră. if (!room.started) { room.players.splice(seat, 1); room.message = `${player.name} a ieșit din cameră.`; // Dacă a ieșit host-ul if (socket.id === room.hostId) { if (room.players.length > 0) { room.hostId = room.players[0].socketId; room.message += ` ${room.players[0].name} este noul host.`; } else { rooms.delete(room.code); continue; } } sendState(room); continue; } // Dacă jocul a început, // îl marcăm doar offline. player.connected = false; room.message = `${player.name} s-a deconectat.`; sendState(room); } });
